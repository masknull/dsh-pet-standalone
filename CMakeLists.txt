# CMakeLists.txt — dsh-pet-standalone (Win32 layered-window desktop pet)
#
# v9 FULL-QUALITY build (the only supported mode): 91 dual-stream VP9 packs
# (main video + alpha-as-gray, staged by scripts/stage-webm2.py, embedded as
# RCDATA) decoded in memory with a statically linked libvpx. Requires libvpx:
#   - MSVC (CI):        vcpkg install libvpx:x64-windows-static
#                       pass the vcpkg toolchain file; VPX is found via
#                       find_path/find_library.
#   - MinGW (local):    VPX_ROOT=<unpacked mingw-w64-x86_64-libvpx dir>
#                       providing include/vpx + lib/libvpx.a + libwinpthread.a
cmake_minimum_required(VERSION 3.16)
project(dsh-pet-standalone CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(VPX_ROOT "" CACHE PATH "dir containing include/vpx and lib/libvpx.a (MinGW/MSYS2 libvpx package)")

add_executable(dsh-pet-standalone WIN32
  src/main.cpp
  src/pet.cpp
  src/anim.cpp
  src/webm.cpp
  src/inflate.cpp
  src/jsonc.cpp
  src/config.cpp
  src/resources.cpp
  src/bubble.cpp
  src/httpserver.cpp
  src/selftest_vectors.h
)

# Generated by scripts/rcgen.py from assets/embedded/ (RCDATA single-exe resources).
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/embedded.rc)
  target_sources(dsh-pet-standalone PRIVATE src/embedded.rc)
else()
  message(WARNING "src/embedded.rc missing — run scripts/rcgen.py (embedded assets) or the exe will have no embedded animations")
endif()

target_include_directories(dsh-pet-standalone PRIVATE src)

# ---- libvpx (VP9 decode) ----
if(MSVC)
  # CI (vcpkg): the official consumption path for this port.
  #   vcpkg install libvpx:x64-windows-static
  #   -> provides find_package(unofficial-libvpx CONFIG REQUIRED)
  find_package(unofficial-libvpx CONFIG REQUIRED)
  target_link_libraries(dsh-pet-standalone PRIVATE unofficial::libvpx::libvpx)
else()
  if(NOT VPX_ROOT)
    message(FATAL_ERROR "VPX_ROOT must point at the unpacked mingw-w64-x86_64-libvpx dir (include/vpx + lib/libvpx.a + lib/libwinpthread.a)")
  endif()
  target_include_directories(dsh-pet-standalone PRIVATE ${VPX_ROOT}/include)
  target_link_libraries(dsh-pet-standalone PRIVATE
    ${VPX_ROOT}/lib/libvpx.a
    ${VPX_ROOT}/lib/libwinpthread.a)
endif()

target_link_libraries(dsh-pet-standalone PRIVATE user32 gdi32 shell32 advapi32 ws2_32 gdiplus)

if(MSVC)
  target_compile_options(dsh-pet-standalone PRIVATE /utf-8 /O2 /W3)
  set_property(TARGET dsh-pet-standalone PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(dsh-pet-standalone PRIVATE -O2 -ffunction-sections -fdata-sections -Wall -Wextra)
  target_link_options(dsh-pet-standalone PRIVATE -Wl,--gc-sections -static -static-libgcc -static-libstdc++)
  if(MINGW)
    target_link_options(dsh-pet-standalone PRIVATE -municode)
    target_compile_definitions(dsh-pet-standalone PRIVATE _WIN32_WINNT=0x0601)
  endif()
endif()
